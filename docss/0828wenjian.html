按行读取

with open("访客列表.txt", encoding="utf8") as fin:
for line in fin:
line = line[:-1]去掉每一行的最后一个元素
line = line.rstrip()
print(line)

写出文件

with open("数字列表.txt", "w", encoding="utf8") as fout:
for i in range(100):
fout.write(str(i) + "\n")
